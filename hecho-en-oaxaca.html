src/services/stellarService.js — Inicialización y health-check Node.js 
// ───────────────────────────────────────────────────────────────
// stellarService.js — Capa de abstracción sobre el SDK de Stellar
// Conecta con Etherfuse (MXNe) y proveedores de off-ramp
// ───────────────────────────────────────────────────────────────
const StellarSdk = require('@stellar/stellar-sdk');
const axios     = require('axios');

// ─── Config ────────────────────────────────────────────────────
const NETWORK = process.env.STELLAR_NETWORK === 'mainnet'
  ? StellarSdk.Networks.PUBLIC
  : StellarSdk.Networks.TESTNET;

const server = new StellarSdk.Horizon.Server(
  process.env.STELLAR_NETWORK === 'mainnet'
    ? 'https://horizon.stellar.org'
    : 'https://horizon-testnet.stellar.org'
);

// Emisor oficial de MXNe en Stellar (Etherfuse)
const MXNE_ISSUER = process.env.MXNE_ISSUER_ADDRESS;
const MXNe = new StellarSdk.Asset('MXNe', MXNE_ISSUER);

// ─── Crear billetera para nuevo productor ──────────────────────
async function crearBilletera(nombreProductor, municipio) {
  // 1. Generar par de llaves criptográficas ed25519
  const par = StellarSdk.Keypair.random();

  // 2. Fondear cuenta con Friendbot (testnet) o cuenta maestra (mainnet)
  if (process.env.STELLAR_NETWORK !== 'mainnet') {
    await axios.get(`https://friendbot.stellar.org?addr=${par.publicKey()}`);
  } else {
    await fondearDesdeMaestra(par.publicKey(), '1.5'); // mínimo XLM
  }

  // 3. Establecer trust line para MXNe
  const cuenta = await server.loadAccount(par.publicKey());
  const tx = new StellarSdk.TransactionBuilder(cuenta, {
    fee: await server.fetchBaseFee(),
    networkPassphrase: NETWORK,
  })
  .addOperation(StellarSdk.Operation.changeTrust({ asset: MXNe }))
  .setTimeout(30)
  .build();

  tx.sign(par);
  await server.submitTransaction(tx);

  // ⚠️  La llave secreta NUNCA se almacena en texto plano.
  // Cifrar con AES-256-GCM usando PIN del usuario antes de persistir.
  return {
    publicKey:  par.publicKey(),
    secretKey:  par.secret(),   // devolver UNA vez al cliente para cifrado en cliente
    nombreProductor,
    municipio,
    createdAt: new Date().toISOString(),
  };
}

// ─── Enviar MXNe a productor (pago incoming) ──────────────────
async function enviarMXNe(destPublicKey, monto, memo) {
  const maestraKeypair = StellarSdk.Keypair.fromSecret(process.env.MAESTRA_SECRET);
  const maestra = await server.loadAccount(maestraKeypair.publicKey());

  const tx = new StellarSdk.TransactionBuilder(maestra, {
    fee: await server.fetchBaseFee(),
    networkPassphrase: NETWORK,
  })
  .addOperation(StellarSdk.Operation.payment({
    destination: destPublicKey,
    asset:       MXNe,
    amount:      monto.toString(),
  }))
  .addMemo(StellarSdk.Memo.text(memo.substring(0, 28)))
  .setTimeout(30)
  .build();

  tx.sign(maestraKeypair);
  return server.submitTransaction(tx);
}

module.exports = { crearBilletera, enviarMXNe };
03 Flujo de Fondos

Del pago global al efectivo en Putla

Recorrido completo de un pago internacional hasta que el productor retira en efectivo o genera rendimientos.

1
Cliente extranjero inicia el pago

Un comprador en EU/Europa paga con USD, EUR o USDC desde su banco o wallet. El importe se convierte a MXNe usando el tipo de cambio de Etherfuse en tiempo real.

💳 Comprador (EU/Europa) 🔁 Etherfuse FX Engine
2
Etherfuse mina MXNe en Stellar

La API de Etherfuse recibe el fondeo en USD/USDC, valida el monto y mina la cantidad equivalente de MXNe en la blockchain de Stellar (<5 segundos de confirmación).

🏦 Etherfuse Mint ⭐ Stellar Network 💱 MXNe Token
3
Backend recibe el pago y acredita al productor

El backend escucha el streaming de pagos en Horizon (SSE). Al detectar el pago al "hot wallet" de la plataforma, envía MXNe a la billetera del productor correspondiente al ID del pedido.

⚙️ Backend API 📡 Horizon Streaming 👛 Billetera Productor
4
Productor decide: Ahorrar o Retirar

Desde la app, el productor (o su hijo) elige qué hacer con los MXNe recibidos. Sin seleccionar nada, el default es mover el 70% a Stablebonds automáticamente.

💰 Guardar en Stablebonds (~9% APY) 🏧 Retirar en Efectivo (SPEI)
5A
Ruta A: Ahorro con rendimientos (Stablebonds)

El backend llama a la API de Etherfuse para mover MXNe del productor a Stablebonds (CETES tokenizados). Los rendimientos se acumulan diariamente y se reflejan en el saldo visible de la app.

📈 Etherfuse Stablebonds 🏛️ CETES (Gobierno MX)
5B
Ruta B: Retiro en efectivo vía SPEI

El backend quema MXNe vía Etherfuse, recibe MXN en la cuenta custodio y llama a AlfredPay/BlindPay para enviar un SPEI a la CLABE del productor o generar un código de retiro OXXO.

🔥 Etherfuse Burn 🔄 BlindPay / AlfredPay 💵 SPEI / OXXO

05 Interfaz de Usuario

Requerimientos del Frontend

Web-App Progressive (PWA) diseñada para operar con poca conectividad y usuarios con poco conocimiento técnico.

📶
Diseño Offline-First
Service Worker + IndexedDB para caché local. Las operaciones se encolan y sincronizan cuando regresa la señal. La app debe funcionar en 2G intermitente.
CRÍTICO
💰
Saldo en "Pesos" — Sin jerga cripto
Mostrar siempre el equivalente en MXN. Nunca mostrar "MXNe", "Stellar" o "blockchain". El productor ve: "Tu saldo: $1,450 pesos" y "Ganaste $12 pesos esta semana".

📱
Mobile-First, Pantalla Pequeña
Optimizado para pantallas de 4-5 pulgadas. Fuentes grandes (mín. 16px). Botones táctiles de al menos 48×48px. Compatible con Android 7+ básico.
CRÍTICO
🔢
Login por PIN de 4 dígitos + SMS OTP
Sin contraseñas complicadas. PIN de 4 dígitos + verificación por SMS (Twilio). Opción de huella biométrica si el dispositivo lo permite.

Inicializar billetera: Productor de Madera en Putla

Snippet completo para registrar a un nuevo productor y crear su billetera en Stellar con trust line de MXNe.

scripts/registrarProductorMadera.js — Ejemplo de uso Node.js 
/**
 * Registrar productor de madera en Putla, Oaxaca
 * Ejecutar: node registrarProductorMadera.js
 *
 * Dependencias:
 *   npm install @stellar/stellar-sdk axios dotenv
 */
require('dotenv').config();
const StellarSdk = require('@stellar/stellar-sdk');
const axios      = require('axios');

// ─── Configuración ─────────────────────────────────────────────
const ES_TESTNET     = process.env.STELLAR_NETWORK !== 'mainnet';
const HORIZON_URL    = ES_TESTNET
  ? 'https://horizon-testnet.stellar.org'
  : 'https://horizon.stellar.org';
const NETWORK_PHRASE = ES_TESTNET
  ? StellarSdk.Networks.TESTNET
  : StellarSdk.Networks.PUBLIC;

const server = new StellarSdk.Horizon.Server(HORIZON_URL);

// Emisor de MXNe (Etherfuse) — reemplazar con dirección real en mainnet
const MXNE_ISSUER = process.env.MXNE_ISSUER
  ?? 'GDUKMGUGDZQK6YHYA5Z6AY2G4XDSZPSZ3SW5UN3ARVMO6QSRDWP5YCZ'; // testnet ejemplo
const MXNe = new StellarSdk.Asset('MXNe', MXNE_ISSUER);

// ─── Datos del productor ────────────────────────────────────────
const PRODUCTOR = {
  nombre:    'Benito Ramírez Cruz',
  municipio: 'Putla Villa de Guerrero',
  estado:    'Oaxaca',
  categoria: 'madera',
  telefono:  '+529511234567',
};

// ─── Función principal ──────────────────────────────────────────
async function registrarProductor(datos) {
  console.log(`\n🌽 Hecho en Oaxaca — Registrando productor de ${datos.categoria}`);
  console.log(`   ${datos.nombre} · ${datos.municipio}, ${datos.estado}\n`);

  // 1. Generar par de llaves criptográficas
  const keypair = StellarSdk.Keypair.random();
  console.log('🔑 Llave pública  :', keypair.publicKey());
  console.log('🔒 Llave secreta  : [CIFRAR ANTES DE GUARDAR]');

  // 2. Fondear cuenta en testnet con Friendbot (gratuito)
  if (ES_TESTNET) {
    console.log('\n⏳ Fondeando cuenta en testnet...');
    const res = await axios.get(
      `https://friendbot.stellar.org?addr=${keypair.publicKey()}`
    );
    console.log('✅ Cuenta fondeada. TX:', res.data.id);
  }

  // 3. Cargar cuenta desde Horizon
  const cuenta = await server.loadAccount(keypair.publicKey());
  console.log('\n💫 Cuenta cargada. Secuencia:', cuenta.sequenceNumber());

  // 4. Construir transacción con trust line hacia MXNe
  console.log('\n⏳ Estableciendo trust line MXNe...');
  const fee = await server.fetchBaseFee();

  const tx = new StellarSdk.TransactionBuilder(cuenta, {
    fee,
    networkPassphrase: NETWORK_PHRASE,
  })
  .addOperation(
    StellarSdk.Operation.changeTrust({
      asset:  MXNe,
      limit: '1000000', // límite máximo en MXNe (1 millón de pesos)
    })
  )
  .addMemo(StellarSdk.Memo.text('HechoEnOaxaca-Madera'))
  .setTimeout(30)
  .build();

  // 5. Firmar y enviar
  tx.sign(keypair);
  const resultado = await server.submitTransaction(tx);

  console.log('✅ Trust line establecida!');
  console.log('   TX Hash:', resultado.hash);

  // 6. Resultado para guardar en base de datos
  const registro = {
    ...datos,
    stellar_address: keypair.publicKey(),
    // ⚠️ NUNCA almacenar secret_key en texto plano en DB
    // Cifrar con AES-256-GCM usando el PIN del usuario:
    // secret_key_encrypted: await cifrarConPIN(keypair.secret(), pin)
    trust_line_mxne: true,
    trust_line_tx:   resultado.hash,
    kyc_nivel:       0,
    created_at:      new Date().toISOString(),
  };

  console.log('\n📦 Registro listo para DB:');
  console.log(JSON.stringify(registro, null, 2));

  return registro;
}

// ─── Ejecutar ───────────────────────────────────────────────────
registrarProductor(PRODUCTOR)
  .then(() => console.log('\n🌽 ¡Productor registrado exitosamente!\n'))
  .catch(err => {
    console.error('\n❌ Error:', err.message);
    if (err.response?.data?.extras) {
      console.error('   Stellar extras:',
        JSON.stringify(err.response.data.extras.result_codes));
    }
    process.exit(1);
  });
